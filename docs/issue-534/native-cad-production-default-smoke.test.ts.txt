// Temporary acceptance harness; retained verbatim with the validation log.
// @vitest-environment node
import { readFileSync, writeFileSync } from 'node:fs';
import { expect, it } from 'vitest';
import { importCad } from '../src/app/components/documents/geometry/cad';
const module = await WebAssembly.compile(readFileSync('vendor/occt-import-js/dist/occt-import-js.wasm'));
const records: unknown[] = [];
const cases = [
  ['test/fixtures/geometry/lod-assembly.step','step',6136],
  ['/tmp/pr556-curved-200.step','step',1001600],
  ['/tmp/pr556-large-bspline.step','step',216],
  ['/tmp/pr556-curved.iges','iges',15050],
  ['test/fixtures/geometry/lod-assembly.brep','brep',6136],
] as const;
const triangles = (model: any) => model.meshes.reduce((sum: number, mesh: any) => sum + mesh.indices.length / 3, 0);
for (const [fixture,format,expectedTriangles] of cases) it(`production adapter coarse/final: ${fixture}`, async () => {
  const source = new Uint8Array(readFileSync(fixture)), levels: number[] = [], coarse: any[] = [];
  const final = await importCad(source, format, module, undefined, true, (level,model) => { levels.push(level); coarse.push(model); });
  expect(levels).toEqual([0]); expect(source.byteLength).toBe(0);
  expect(triangles(final)).toBe(expectedTriangles);
  expect(coarse[0].root).toEqual(final.root);
  expect(coarse[0].meshes.map((mesh: any) => mesh.name)).toEqual(final.meshes.map(mesh => mesh.name));
  expect(triangles(coarse[0])).toBeLessThan(triangles(final));
  records.push({ fixture, format, levels, meshes: final.meshes.length, coarseTriangles: triangles(coarse[0]), finalTriangles: triangles(final), ownedSourceReleased: source.byteLength === 0 });
  writeFileSync('/tmp/pr556-native-production-default-smoke.json', JSON.stringify(records,null,2)+'\n');
},60000);
